        // --- 2. CONFIGURATION DU SCANNER ---
        // Initialisation de l'objet de scan lié au <div> "reader"
        const html5QrCode = new Html5Qrcode("reader");

        // Fonction exécutée lorsqu'un code est lu avec succès
        function onScanSuccess(decodedText, decodedResult) {
            // Mettre le scanner en pause pour éviter les doubles scans instantanés
            html5QrCode.pause(); 

            // Chercher le code scanné dans le catalogue
            let designation = catalogue[decodedText];
            
            if (designation) {
                // Si trouvé, demander la quantité via une boîte de dialogue
                let qty = prompt("Matériel reconnu : " + designation + "\n\nEntrez la quantité :");
                
                // Si la quantité est valide, on l'ajoute au tableau
                if (qty !== null && qty !== "" && !isNaN(qty)) {
                    let tbody = document.querySelector("#table-inventaire tbody");
                    let row = tbody.insertRow(0); // Insérer la nouvelle ligne en haut
                    row.insertCell(0).innerText = decodedText;
                    row.insertCell(1).innerText = designation;
                    row.insertCell(2).innerText = qty;
                }
            } else {
                // Si le code n'est pas dans le catalogue
                alert("Ce code (" + decodedText + ") n'est pas répertorié dans le catalogue.");
            }
            
            // Relancer le scanner automatiquement après 1,5 seconde
            setTimeout(() => html5QrCode.resume(), 1500);
        }

        // --- 3. DÉMARRAGE DU SCANNER ---
        // Commande stricte demandant la caméra arrière (facingMode: "environment")
        // La librairie html5-qrcode gère automatiquement la demande d'autorisation de caméra.
        html5QrCode.start(
            { facingMode: "environment" }, // Demande la caméra principale (dos)
            { 
                fps: 10,    // Nombre d'images par seconde pour l'analyse
                qrbox: {width: 250, height: 150} // Zone de visée à l'écran
            },
            onScanSuccess // Fonction à appeler en cas de succès
        ).catch(err => {
            // En cas d'erreur de caméra (refus, pas de caméra...)
            alert("Erreur lors du démarrage de la caméra arrière : " + err);
        });

        // --- 4. FONCTION D'EXPORTATION CORRIGÉE (FIX Â², NOM DYNAMIQUE) ---
        function exporterCSV() {
            // A. Logique : Nom de fichier dynamique avec date du jour (jj_mm_aaaa)
            const maintenant = new Date();
            const jour = String(maintenant.getDate()).padStart(2, '0');
            const mois = String(maintenant.getMonth() + 1).padStart(2, '0'); // Janvier est 0!
            const annee = maintenant.getFullYear();
            // J'ai remplacé les '/' demandés par des '_' car '/' est interdit dans les noms de fichiers.
            const nomFichier = `sortie_de_stock_${jour}_${mois}_${annee}.csv`;

            // B. Logique : Construction du contenu texte du CSV
            
            // 1. Instruction STRICTE "sep=;" en première ligne pour forcer Excel à créer les colonnes
            let csvStringContent = "sep=;\nCode Barre;Designation;Quantite\n";
            
            // 2. Récupérer les données du tableau d'inventaire
            let rows = document.querySelectorAll("#table-inventaire tbody tr");
            
            if(rows.length === 0) {
                alert("La liste est vide. Scannez des articles avant d'exporter.");
                return;
            }

            rows.forEach(row => {
                let cols = row.querySelectorAll("td");
                // On s'assure de ne pas avoir de points-virgules dans les textes pour ne pas décaler les colonnes
                let code = cols[0].innerText.replace(/;/g, "");
                let designation = cols[1].innerText.replace(/;/g, "");
                let quantite = cols[2].innerText.replace(/;/g, "");
                
                csvStringContent += code + ";" + designation + ";" + quantite + "\n";
            });

            // C. Logique : Encodage STRICT en Windows-1252 (mono-octet) pour résoudre le bug Â²
            
            // 1. Fonction pour convertir une chaîne JavaScript (multi-octets) en octets bruts Windows-1252 (mono-octet)
            // Cela transforme le caractère '²' en l'octet unique 0xB2, que Excel interprète correctement,
            // bypassing ainsi l'interprétation UTF-8 erronée (qui décompose '²' en deux octets 0xC2 0xB2 interprétés Â²).
            // Le même fix s'applique pour les accents standards Ouest-Européens.
            function stringToWindows1252Bytes(str) {
                const bytes = new Uint8Array(str.length);
                for (let i = 0; i < str.length; i++) {
                    const charCode = str.charCodeAt(i);
                    // Le caractère superscript '²' (Unicode code 178 / 0x00B2)
                    // se mappe directement sur l'octet 178 (0xB2) en Windows-1252.
                    if (charCode <= 255) {
                        bytes[i] = charCode;
                    } else {
                        // Caractère non gérable en Windows-1252 (rare pour cette application), remplacé par un placeholder.
                        bytes[i] = '?'.charCodeAt(0);
                    }
                }
                return bytes;
            }

            // 2. Encoder le contenu complet du CSV en tableau d'octets bruts Windows-1252
            const encodedDataRaw = stringToWindows1252Bytes(csvStringContent);

            // D. Logique : Création et téléchargement du fichier Blob final
            
            // Création du Blob à partir du tableau d'octets directs (Plus besoin de charset ici)
            let blob = new Blob([encodedDataRaw], { type: "text/csv" });
            
            // Générer un lien invisible pour lancer le téléchargement
            let link = document.createElement("a");
            link.href = URL.createObjectURL(blob);
            link.download = nomFichier; // Utilise le nom dynamique : sortie_de_stock_jj_mm_aaaa.csv
            link.click();
        }
